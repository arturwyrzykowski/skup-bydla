# Odcinek 21 „Kulig” — plan wykonania

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zbudować i opublikować odcinek 21 „Kulig” (`bajka/kulig.html`) wg `docs/superpowers/specs/2026-10-04-kulig-design.md` — z montażem ujęć, zbliżeniami i stylem Tarantino.

**Architecture:** Jeden samodzielny plik HTML (SVG 1600×900 + Web Animations API + WebAudio), składany skryptem `build_ep21.py` z małych modułów: szkielet strony i silnik bierzemy z aktualnego `nogi.html` (ma już odtwarzacz nagrań z `fix_audio.py`, `flap()`, ukryte dymki); nowe są: silnik ujęć (`ep21_kino.js`), dźwięk (`ep21_audio.js`), postacie zimowe (`ep21_postacie.py`), ujęcia rozdziałów (`ep21_sceny.py`), scenariusz (`ep21_play.js`). Głosy: `dubbing.py`, potem `fix_audio.py`.

**Tech Stack:** Python 3 (builder), SVG/JS w przeglądarce, headless Chrome do testów (`dub_capture.py` — `FAST`, `AUTO`, `capture()`), ElevenLabs (darmowe głosy).

```js
  // ===== KINO: ujecia, kamera, pasy, plansze, ziarno =====
  const SHOTS = [...document.querySelectorAll("#shots > g")];
  function shot(id, cam = { x: 800, y: 450, k: 1 }){
    SHOTS.forEach(g => g.style.display = g.id === id ? "" : "none");
    camSet(cam);
  }
  // kamera = transform calego #shots (punkt (x,y) na srodku ekranu, powiekszenie k)
  const camT = c => `translate(800px,450px) scale(${c.k}) translate(${-c.x}px,${-c.y}px)`;
  let cam = { x: 800, y: 450, k: 1 };
  function camSet(c){ cam = { ...c }; $("shots").getAnimations().forEach(a => a.cancel()); $("shots").style.transform = camT(cam); }
  async function camTo(c, ms, easing = "ease-in-out"){
    const to = { ...cam, ...c };
    await anim($("shots"), [{ transform: camT(cam) }, { transform: camT(to) }], { duration: ms, easing });
    camSet(to);
  }
  const crash = (c, ms = 160) => camTo(c, ms, "cubic-bezier(.2,.9,.2,1)");   // szybki najazd
  function shake(power = 12, ms = 400){
    return anim($("cam"), [0, 1, 2, 3, 4, 5].map(i => ({ transform: i === 5 ? "translate(0,0)" :
      `translate(${(Math.random() - .5) * power * 2}px,${(Math.random() - .5) * power * 2}px)` })), { duration: ms });
  }
  // plansza rozdzialu (czarne tlo, zolty napis jak u Tarantino)
  async function chapter(nr, title, ms = 2200){
    $("chNr").textContent = "ROZDZIAŁ " + nr; $("chTitle").textContent = title;
    $("chapter").style.display = ""; sfx.chapter && sfx.chapter();
    await anim($("chapter"), [{ opacity: 0 }, { opacity: 1 }], { duration: 300 });
    await sleep(ms);
    await anim($("chapter"), [{ opacity: 1 }, { opacity: 0 }], { duration: 300 });
    $("chapter").style.display = "none";
  }
  // zatrzymana klatka: wszystko staje, lekki zoom + biale mrugniecie
  async function freeze(ms = 1200){
    const all = document.getAnimations(); all.forEach(a => a.pause());
    $("flash").style.opacity = 0.8; await sleep(60); $("flash").style.opacity = 0;
    await sleep(ms); all.forEach(a => { try { a.play(); } catch (e) {} });
  }
  // glitch (zawieszony obraz): przesuniecia kolorow i paski
  async function glitch(ms = 1800){
    $("glitch").style.display = "";
    const t0 = performance.now();
    while (performance.now() - t0 < ms){
      $("cam").style.transform = `translate(${(Math.random() - .5) * 40}px,0)`;
      $("glitch").setAttribute("y", Math.random() * 800);
      $("glitch").setAttribute("height", 20 + Math.random() * 120);
      await sleep(70);
    }
    $("cam").style.transform = ""; $("glitch").style.display = "none";
  }
  // snieg: platki spadaja w petli w warstwie #snow (wlaczany per ujecie)
  function snow(on, n = 90){
    $("snow").innerHTML = ""; $("snow").style.display = on ? "" : "none";
    if (!on) return;
    for (let i = 0; i < n; i++){
      const c = document.createElementNS(NS, "circle"), x = Math.random() * 1600, r = 2 + Math.random() * 4;
      c.setAttribute("cx", x); c.setAttribute("cy", -20); c.setAttribute("r", r); c.setAttribute("fill", "#fff"); c.setAttribute("opacity", 0.85);
      $("snow").append(c);
      c.animate([{ transform: "translate(0,0)" }, { transform: `translate(${(Math.random() - .5) * 160}px,940px)` }],
        { duration: 4000 + Math.random() * 5000, delay: -Math.random() * 9000, iterations: Infinity });
    }
  }
  // para z ust / dym: oblok rosnie i znika (x,y w ukladzie aktualnego ujecia -> warstwa #fx2 w #shots)
  function puff(x, y, { col = "#fff", r = 14, up = 60, ms = 1400, op = 0.7 } = {}){
    const c = document.createElementNS(NS, "circle");
    c.setAttribute("cx", x); c.setAttribute("cy", y); c.setAttribute("r", r); c.setAttribute("fill", col);
    $("fx2").append(c);
    c.animate([{ transform: "translate(0,0) scale(.4)", opacity: op }, { transform: `translate(${(Math.random() - .5) * 40}px,${-up}px) scale(2.2)`, opacity: 0 }],
      { duration: ms, easing: "ease-out" }).finished.then(() => c.remove()).catch(() => {});
  }
  function breath(x, y){ puff(x, y, { r: 8, up: 30, ms: 900, op: 0.55 }); }
  function smoke(x, y, n = 6, col = "#2a2a2a"){ for (let i = 0; i < n; i++) setTimeout(() => puff(x, y, { col, r: 20, up: 160, ms: 1800, op: 0.85 }), i * 120); }
```
